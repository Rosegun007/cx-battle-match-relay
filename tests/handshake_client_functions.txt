      function startOnlineClockProbe(generation=online.connectionGeneration){
        if(generation!==online.connectionGeneration)return;
        online.probeCount=0;online.bestClockRtt=Infinity;
        online.recentRtts=[];online.relayRttEstimate=Infinity;
        online.probeComplete=false;
      }
      function replyServerClockProbeS11(data,generation=online.connectionGeneration){
        if(generation!==online.connectionGeneration||online.ws?.readyState!==WebSocket.OPEN)return;
        if(!Number.isInteger(data.seq)||data.seq!==online.probeCount+1||data.seq>3||typeof data.nonce!=='string')return;
        const at=Date.now();
        onlineSend({op:'clock_probe_reply',seq:data.seq,nonce:data.nonce,clientReceivedAt:at,clientSentAt:Date.now()});
      }
      function handleServerClockResultS11(data,generation=online.connectionGeneration){
        if(generation!==online.connectionGeneration||data.seq!==online.probeCount+1||!Number.isFinite(data.rttMs))return;
        const {serverSentAt:t0,clientReceivedAt:t1,clientSentAt:t2,serverReceivedAt:t3}=data;
        if(![t0,t1,t2,t3].every(Number.isFinite))return;
        const offset=((t0-t1)+(t3-t2))/2;
        const rtt=Math.max(0,data.rttMs);
        online.recentRtts.push(rtt);
        online.relayRttEstimate=Math.max(...online.recentRtts);
        if(rtt<online.bestClockRtt){online.bestClockRtt=rtt;online.serverClockOffsetMs=offset;}
        online.probeCount++;
      }
      function completeServerClockProbeS11(data,generation=online.connectionGeneration){
        if(generation!==online.connectionGeneration||online.probeCount!==3||data.samples!==3||online.probeComplete)return;
        online.probeComplete=true;
        // Do not start regular heartbeats before the 3-step server clock calibration is complete.
        startHeartbeatTimerS8(generation);
        joinOnlineQueue(generation);
      }
      function joinOnlineQueue(generation=online.connectionGeneration) {
        if (generation !== online.connectionGeneration || !online.ws || online.ws.readyState !== WebSocket.OPEN) return;
        onlineSend({
          op:'join_queue',
          protocol:2, wireProtocol:CX_WIRE_20361, simulationProtocol:CX_SIMULATION_439, tickRate:30,
          game:'cx-battle',
          ruleVersion:CX_RULE_VERSION,
          deploySpeedTierCount:liveDeploySpeedTierCount,
          clientId:`cx_${crypto.randomUUID().slice(0,8)}`,
          rttMs:Number.isFinite(online.relayRttEstimate)
            ? online.relayRttEstimate
            : (Number.isFinite(online.bestClockRtt) ? online.bestClockRtt : null)
        });
      }
